#---tcp_client-----

import socket

sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.connect(("localhost", 1111))

while True:
  # Type message for server
  message = input("Enter message for the Server: ")
  sock.send(message.encode())

  # Receive response from server
  response = sock.recv(1024).decode().strip()
  print(f"Server: {response}")

  if response == "goodbye" or message == "goodbye":
    break

sock.close()
print("Server closed")

#------tcp_server-----

import socket

server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server_socket.bind(("localhost", 1111))
server_socket.listen(1)
print("Server is running and waiting for client connection...")
client_socket, addr = server_socket.accept()
print("Client connected!")

while True:
  # Receive message from client
  message = client_socket.recv(1024).decode().strip()
  print(f"Client: {message}")
  if message == "goodbye":
    break
  # Type response for client
  resp = input("Enter Msg for Client: ")
  client_socket.send(resp.encode())

  if resp == "goodbye":
    break

client_socket.close()
server_socket.close()
print("Client closed")
